# Session-Scoped Terminal Integration Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在 session 详情页加 ephemeral 终端：chat topbar 右上一个 toggle 按钮，点开 → spawn PTY 替换 chat 视图、cwd 落在 session 的工作目录；再点 → kill PTY。本地 session 用 `creack/pty`，远端 session 用 agentred 的 PTY，前端对底层透明。

**Architecture:** 三层。`internal/pkg/pty` 定义 `Handle` interface（运行时 PTY API）和 `local` / `remote` 两个 impl；`internal/service/terminal_svc` 暴露单一 Wails service，内部 `PTYBackend` factory 接口根据 `session.Backend.DeviceID` 选 backend；`internal/daemon/handlers/terminal.go` 在 agentred 上跑实际远端 PTY。前端 `terminal-panel.tsx` 用 xterm.js + 单一 Wails service + namespaced events `terminal:<sessionID>:data|exit`。

**Tech Stack:** Go 1.26（goconvey + sqlmock + mockgen）、Wails v2、`github.com/creack/pty`、React 19 + TypeScript、Vitest、`@xterm/xterm` + `@xterm/addon-fit` + `@xterm/addon-web-links`、zustand。

**Spec:** `docs/superpowers/specs/2026-05-28-terminal-integration-design.md`

**Mockup:** `agentry.pen` → frame `Terminal Option C — Top Tab Switch` (id `qRiqR`)（最接近，实际无 tab strip，仅 topbar toggle 按钮）。

```go
// Package pty defines a backend-agnostic interface to an interactive
// pseudo-terminal. Implementations live under pty/local (creack/pty) and
// pty/remote (agentred ws proxy).
package pty

import "context"

// ExitReason 描述 PTY 退出的语义来源。
type ExitReason string

const (
	ExitNatural        ExitReason = "natural"          // shell 自己 exit
	ExitKilled         ExitReason = "killed"           // 被 Close() 主动杀
	ExitConnectionLost ExitReason = "connection_lost" // 远端 ws 断
	ExitDaemonShutdown ExitReason = "daemon_shutdown" // agentred 退出
	ExitError          ExitReason = "error"           // spawn/runtime 错误
)

// ExitInfo 走 Exit() chan 推给消费者，map 清理也走它。
type ExitInfo struct {
	Code   int
	Reason ExitReason
	Msg    string
}

// Handle 是一个已存在 PTY 实例的运行时 API。所有方法都必须可重入安全
// (多次 Close 不能 panic)。Data/Exit chan 在 Close 之后必须最终关闭，
// 这样消费者 select 不会泄露。
type Handle interface {
	Write(data []byte) error
	Resize(cols, rows uint16) error
	Close(ctx context.Context) error
	Data() <-chan []byte
	Exit() <-chan ExitInfo
}
```

```go
//go:build !windows

// Package local implements pty.Handle on top of github.com/creack/pty
// using the host's /bin/sh or $SHELL. Test scope only covers Unix.
package local

import (
	"context"
	"errors"
	"io"
	"os"
	"os/exec"
	"sync"
	"syscall"
	"time"

	cpty "github.com/creack/pty"

	"agentre/internal/pkg/pty"
)

// ErrClosed is returned from Write/Resize after Close.
var ErrClosed = errors.New("pty: closed")

// Config 是 spawn 的输入。Env 已经是 KEY=VAL 数组形式。
type Config struct {
	Cwd, Shell string
	Env        []string
	Cols, Rows uint16
}

type handle struct {
	cmd  *exec.Cmd
	ptmx *os.File
	data chan []byte
	exit chan pty.ExitInfo

	closeOnce sync.Once
	closed    chan struct{}
	closedFlag bool
	mu         sync.Mutex // guards closedFlag
}

// Open spawns the shell under a fresh PTY. Caller must consume Data() or it
// will block writers eventually (8KB chunks * channel buffer 64).
func Open(ctx context.Context, cfg Config) (pty.Handle, error) {
	cmd := exec.CommandContext(ctx, cfg.Shell)
	cmd.Dir = cfg.Cwd
	cmd.Env = cfg.Env

	ws := &cpty.Winsize{Cols: cfg.Cols, Rows: cfg.Rows}
	ptmx, err := cpty.StartWithSize(cmd, ws)
	if err != nil {
		return nil, err
	}

	h := &handle{
		cmd:    cmd,
		ptmx:   ptmx,
		data:   make(chan []byte, 64),
		exit:   make(chan pty.ExitInfo, 1),
		closed: make(chan struct{}),
	}

	go h.readLoop()
	go h.reapLoop()
	return h, nil
}

func (h *handle) readLoop() {
	buf := make([]byte, 8192)
	for {
		n, err := h.ptmx.Read(buf)
		if n > 0 {
			chunk := make([]byte, n)
			copy(chunk, buf[:n])
			select {
			case h.data <- chunk:
			case <-h.closed:
				return
			}
		}
		if err != nil {
			return
		}
	}
}

func (h *handle) reapLoop() {
	err := h.cmd.Wait()
	h.mu.Lock()
	killed := h.closedFlag
	h.mu.Unlock()

	info := pty.ExitInfo{}
	if killed {
		info.Reason = pty.ExitKilled
		info.Code = -1
	} else {
		info.Reason = pty.ExitNatural
		if exitErr, ok := err.(*exec.ExitError); ok {
			info.Code = exitErr.ExitCode()
		}
	}
	select {
	case h.exit <- info:
	default:
	}
	close(h.exit)
	h.markClosed()
	_ = h.ptmx.Close()
	close(h.data)
}

func (h *handle) markClosed() {
	h.closeOnce.Do(func() {
		h.mu.Lock()
		h.closedFlag = true
		h.mu.Unlock()
		close(h.closed)
	})
}

func (h *handle) Data() <-chan []byte         { return h.data }
func (h *handle) Exit() <-chan pty.ExitInfo   { return h.exit }

func (h *handle) Write(data []byte) error {
	h.mu.Lock()
	if h.closedFlag {
		h.mu.Unlock()
		return ErrClosed
	}
	h.mu.Unlock()
	_, err := h.ptmx.Write(data)
	if err != nil && errors.Is(err, io.ErrClosedPipe) {
		return ErrClosed
	}
	return err
}

func (h *handle) Resize(cols, rows uint16) error {
	h.mu.Lock()
	if h.closedFlag {
		h.mu.Unlock()
		return ErrClosed
	}
	h.mu.Unlock()
	return cpty.Setsize(h.ptmx, &cpty.Winsize{Cols: cols, Rows: rows})
}

// Close SIGHUPs the shell, gives it 200ms, then SIGKILLs the process group.
func (h *handle) Close(ctx context.Context) error {
	h.mu.Lock()
	if h.closedFlag {
		h.mu.Unlock()
		return nil
	}
	h.closedFlag = true
	h.mu.Unlock()

	pid := h.cmd.Process.Pid
	_ = syscall.Kill(-pid, syscall.SIGHUP) // process group
	select {
	case <-time.After(200 * time.Millisecond):
		_ = syscall.Kill(-pid, syscall.SIGKILL)
	case <-ctx.Done():
		_ = syscall.Kill(-pid, syscall.SIGKILL)
		return ctx.Err()
	}
	h.markClosed()
	return nil
}
```

- [ ] **Step 2: Note process-group requirement**

`exec.Cmd` 默认进程组共享父 pid；creack/pty 的 `StartWithSize` 在 Unix 上隐式 setsid（子进程是 session leader、自己的 pgid），所以 `kill(-pid, ...)` 能命中 shell 及其后代。Windows 上不适用——package build tag 已经排除。

```go
package terminal_svc

import (
	"context"
	"fmt"
)

// Emitter abstracts wails runtime.EventsEmit so service tests can record
// emissions. Production wiring is in internal/app/terminal.go.
type Emitter interface {
	Emit(ctx context.Context, name string, payload any)
}

type EmitterFunc func(ctx context.Context, name string, payload any)

func (f EmitterFunc) Emit(ctx context.Context, name string, payload any) {
	if f != nil {
		f(ctx, name, payload)
	}
}

type NoopEmitter struct{}

func (NoopEmitter) Emit(context.Context, string, any) {}

// DataEvent payload pushed to terminal:<sid>:data.
type DataEvent struct {
	Data string `json:"data"`
}

// ExitEvent payload pushed to terminal:<sid>:exit.
type ExitEvent struct {
	Code   int    `json:"code"`
	Reason string `json:"reason"`
	Msg    string `json:"msg"`
}

func DataEventName(sessionID int64) string {
	return fmt.Sprintf("terminal:%d:data", sessionID)
}

func ExitEventName(sessionID int64) string {
	return fmt.Sprintf("terminal:%d:exit", sessionID)
}
```

```go
	defer r.mu.Unlock()
	out := make([]recordedEmit, len(r.emit))

	return out
```

```go
package terminal_svc

import (
	"context"
	"errors"
	"sync"

	"agentre/internal/pkg/pty"
)

// SessionLookup resolves a sessionID to its working directory. Implementations
// wrap chat_svc.ResolveSessionCwd in production.
type SessionLookup interface {
	ResolveCwd(ctx context.Context, sessionID int64) (string, error)
}

// ErrTerminalClosed indicates the sessionID has no live PTY (never opened,
// already exited, or already closed).
var ErrTerminalClosed = errors.New("terminal_svc: terminal closed or never opened")

// Service is the Wails-bound entrypoint. One instance per process; exposed
// to the frontend through methods on internal/app/App.
type Service struct {
	backend PTYBackend
	lookup  SessionLookup
	emitter Emitter

	mu       sync.Mutex
	handles  map[int64]pty.Handle
	cancels  map[int64]context.CancelFunc
}

func New(backend PTYBackend, lookup SessionLookup, emitter Emitter) *Service {
	if emitter == nil {
		emitter = NoopEmitter{}
	}
	return &Service{
		backend: backend, lookup: lookup, emitter: emitter,
		handles: map[int64]pty.Handle{},
		cancels: map[int64]context.CancelFunc{},
	}
}

func (s *Service) Open(ctx context.Context, sessionID int64, cols, rows uint16) error {
	cwd, err := s.lookup.ResolveCwd(ctx, sessionID)
	if err != nil {
		return err
	}

	s.closeExisting(sessionID)

	spawnCtx, cancel := context.WithCancel(context.Background())
	h, err := s.backend.Open(spawnCtx, SpawnRequest{
		SessionID: sessionID, Cwd: cwd, Cols: cols, Rows: rows,
	})
	if err != nil {
		cancel()
		return err
	}

	s.mu.Lock()
	s.handles[sessionID] = h
	s.cancels[sessionID] = cancel
	s.mu.Unlock()

	go s.pumpData(sessionID, h)
	go s.pumpExit(sessionID, h)
	return nil
}

func (s *Service) Write(_ context.Context, sessionID int64, data string) error {
	s.mu.Lock()
	h, ok := s.handles[sessionID]
	s.mu.Unlock()
	if !ok {
		return ErrTerminalClosed
	}
	return h.Write([]byte(data))
}

func (s *Service) Resize(_ context.Context, sessionID int64, cols, rows uint16) error {
	s.mu.Lock()
	h, ok := s.handles[sessionID]
	s.mu.Unlock()
	if !ok {
		return ErrTerminalClosed
	}
	return h.Resize(cols, rows)
}

func (s *Service) Close(ctx context.Context, sessionID int64) error {
	s.mu.Lock()
	h, ok := s.handles[sessionID]
	cancel := s.cancels[sessionID]
	if ok {
		delete(s.handles, sessionID)
		delete(s.cancels, sessionID)
	}
	s.mu.Unlock()
	if !ok {
		return nil
	}
	if cancel != nil {
		cancel()
	}
	return h.Close(ctx)
}

// Stop is called from app Shutdown — Close every live PTY, do not block app exit.
func (s *Service) Stop(ctx context.Context) {
	s.mu.Lock()
	ids := make([]int64, 0, len(s.handles))
	for id := range s.handles {
		ids = append(ids, id)
	}
	s.mu.Unlock()
	for _, id := range ids {
		_ = s.Close(ctx, id)
	}
}

func (s *Service) closeExisting(sessionID int64) {
	s.mu.Lock()
	h, ok := s.handles[sessionID]
	cancel := s.cancels[sessionID]
	if ok {
		delete(s.handles, sessionID)
		delete(s.cancels, sessionID)
	}
	s.mu.Unlock()
	if !ok {
		return
	}
	if cancel != nil {
		cancel()
	}
	_ = h.Close(context.Background())
}

func (s *Service) pumpData(sessionID int64, h pty.Handle) {
	name := DataEventName(sessionID)
	for chunk := range h.Data() {
		s.emitter.Emit(context.Background(), name, DataEvent{Data: string(chunk)})
	}
}

func (s *Service) pumpExit(sessionID int64, h pty.Handle) {
	info, ok := <-h.Exit()
	s.mu.Lock()
	delete(s.handles, sessionID)
	if c := s.cancels[sessionID]; c != nil {
		c()
		delete(s.cancels, sessionID)
	}
	s.mu.Unlock()
	if !ok {
		info = pty.ExitInfo{Reason: pty.ExitError, Msg: "exit channel closed"}
	}
	s.emitter.Emit(context.Background(), ExitEventName(sessionID), ExitEvent{
		Code: info.Code, Reason: string(info.Reason), Msg: info.Msg,
	})
}
```

```ts
import { useCallback, useEffect } from "react";
import {
  TerminalOpen,
  TerminalWrite,
  TerminalResize,
  TerminalClose,
} from "@/wailsjs/go/app/App";
import { EventsOn, EventsOff } from "@/wailsjs/runtime/runtime";

interface UseTerminalProps {
  sessionId: number;
  cols: number;
  rows: number;
  onData?: (chunk: string) => void;
  onExit?: (info: { code: number; reason: string; msg: string }) => void;
}

interface UseTerminalApi {
  write: (data: string) => Promise<void>;
  resize: (cols: number, rows: number) => Promise<void>;
  close: () => Promise<void>;
}

export function useTerminal(props: UseTerminalProps): UseTerminalApi {
  const { sessionId, cols, rows, onData, onExit } = props;

  useEffect(() => {
    let aborted = false;
    const dataName = `terminal:${sessionId}:data`;
    const exitName = `terminal:${sessionId}:exit`;

    EventsOn(dataName, (p: { data: string }) => { onData?.(p.data); });
    EventsOn(exitName, (p: { code: number; reason: string; msg: string }) => {
      onExit?.(p);
    });

    TerminalOpen(sessionId, cols, rows).catch((err: unknown) => {
      if (!aborted) onExit?.({ code: -1, reason: "error", msg: String(err) });
    });

    return () => {
      aborted = true;
      EventsOff(dataName);
      EventsOff(exitName);
      void TerminalClose(sessionId);
    };
    // sessionId intentionally only re-binds on identity change
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [sessionId]);

  const write = useCallback((data: string) => TerminalWrite(sessionId, data), [sessionId]);
  const resize = useCallback((c: number, r: number) => TerminalResize(sessionId, c, r), [sessionId]);
  const close = useCallback(() => TerminalClose(sessionId), [sessionId]);

  return { write, resize, close };
}
```

```tsx
import { useEffect, useRef, useState } from "react";
import { Terminal as XTerm } from "@xterm/xterm";
import { FitAddon } from "@xterm/addon-fit";
import { WebLinksAddon } from "@xterm/addon-web-links";
import "@xterm/xterm/css/xterm.css";

import { useTerminal } from "./use-terminal";

interface TerminalPanelProps {
  sessionId: number;
}

interface ExitState {
  code: number;
  reason: string;
  msg: string;
}

export function TerminalPanel({ sessionId }: TerminalPanelProps) {
  const containerRef = useRef<HTMLDivElement | null>(null);
  const termRef = useRef<XTerm | null>(null);
  const fitRef = useRef<FitAddon | null>(null);
  const [exit, setExit] = useState<ExitState | null>(null);

  const hook = useTerminal({
    sessionId,
    cols: 80,
    rows: 24,
    onData: (chunk) => termRef.current?.write(chunk),
    onExit: (info) => setExit(info),
  });

  useEffect(() => {
    if (!containerRef.current) return;
    const term = new XTerm({
      fontFamily: "'JetBrains Mono', 'Menlo', 'Monaco', monospace",
      fontSize: 13,
      theme: { background: "#0b1220", foreground: "#e2e8f0" },
      scrollback: 500,
      convertEol: true,
    });
    const fit = new FitAddon();
    term.loadAddon(fit);
    term.loadAddon(new WebLinksAddon());
    term.open(containerRef.current);
    fit.fit();
    term.onData((d) => { void hook.write(d); });
    term.onResize(({ cols, rows }) => { void hook.resize(cols, rows); });
    termRef.current = term;
    fitRef.current = fit;

    const onResize = () => fit.fit();
    window.addEventListener("resize", onResize);
    return () => {
      window.removeEventListener("resize", onResize);
      term.dispose();
      termRef.current = null;
      fitRef.current = null;
    };
    // term lifecycle is sessionId-scoped
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [sessionId]);

  return (
    <div
      data-testid="terminal-panel"
      className="relative h-full w-full bg-[#0b1220] p-2"
    >
      <div ref={containerRef} className="h-full w-full" />
      {exit && exit.code !== 0 && (
        <div className="absolute bottom-2 left-2 right-2 rounded bg-red-900/80 px-3 py-1 text-xs text-red-100">
          shell 退出 (code {exit.code}, {exit.reason})
        </div>
      )}
    </div>
  );
}
```

`xterm` 默认 Ctrl+C 直接发 SIGINT，但浏览器原生行为是"有选区时复制"。加 `attachCustomKeyEventHandler`：

```tsx
// inside useEffect, after term.open():
term.attachCustomKeyEventHandler((ev) => {
  if (ev.type !== "keydown") return true;
  const meta = ev.metaKey || ev.ctrlKey;
  if (meta && ev.key === "c" && term.hasSelection()) {
    void navigator.clipboard.writeText(term.getSelection());
    return false; // 阻止 xterm 处理（不发 SIGINT）
  }
  if (meta && ev.key === "v") {
    void navigator.clipboard.readText().then((t) => { void hook.write(t); });
    return false;
  }
  return true; // 其余交给 xterm
});
```

```go
// TerminalOpenParams is the request for terminal.open.
type TerminalOpenParams struct {
	SessionID  int64    `json:"sessionId"`
	Cwd        string   `json:"cwd"`
	Shell      string   `json:"shell"`
	Env        []string `json:"env"`
	Cols, Rows uint16   `json:"cols,rows"`
}

// TerminalOpenResult is the response.
type TerminalOpenResult struct {
	TerminalID string `json:"terminalId"`
}

type TerminalWriteParams struct {
	TerminalID string `json:"terminalId"`
	Data       string `json:"data"`
}

type TerminalResizeParams struct {
	TerminalID string `json:"terminalId"`
	Cols, Rows uint16 `json:"cols,rows"`
}

type TerminalCloseParams struct {
	TerminalID string `json:"terminalId"`
}

// TerminalDataEvent is pushed daemon→client on stdout chunks.
type TerminalDataEvent struct {
	TerminalID string `json:"terminalId"`
	Data       string `json:"data"`
}

// TerminalExitEvent is pushed when the remote PTY exits.
type TerminalExitEvent struct {
	TerminalID string `json:"terminalId"`
	Code       int    `json:"code"`
	Reason     string `json:"reason"`
	Msg        string `json:"msg"`
}
```

```go
import (
	"context"
	"encoding/json"
	"sync"

	"github.com/google/uuid"

	"agentre/internal/pkg/pty"
	"agentre/internal/pkg/pty/local"
)

// Pusher abstracts the ws push mechanism for tests.
type Pusher interface {
	Push(event string, body any)
}

type TerminalHandlers struct {
	push    Pusher
	mu      sync.Mutex
	handles map[string]pty.Handle
}

func NewTerminalHandlers(p Pusher) *TerminalHandlers {
	return &TerminalHandlers{push: p, handles: map[string]pty.Handle{}}
}

func (h *TerminalHandlers) Open(ctx context.Context, raw json.RawMessage) (any, error) {
	var p TerminalOpenParams
	if err := json.Unmarshal(raw, &p); err != nil {
		return nil, err
	}
	if p.Shell == "" {
		p.Shell = "/bin/sh"
	}
	handle, err := local.Open(ctx, local.Config{
		Cwd: p.Cwd, Shell: p.Shell, Env: p.Env,
		Cols: p.Cols, Rows: p.Rows,
	})
	if err != nil {
		return nil, err
	}
	tid := uuid.NewString()
	h.mu.Lock()
	h.handles[tid] = handle
	h.mu.Unlock()

	go h.pumpData(tid, handle)
	go h.pumpExit(tid, handle)
	return TerminalOpenResult{TerminalID: tid}, nil
}

func (h *TerminalHandlers) Write(_ context.Context, raw json.RawMessage) (any, error) {
	var p TerminalWriteParams
	if err := json.Unmarshal(raw, &p); err != nil {
		return nil, err
	}
	h.mu.Lock()
	handle, ok := h.handles[p.TerminalID]
	h.mu.Unlock()
	if !ok {
		return nil, ErrSessionNotFound // 复用现有 error 表
	}
	return struct{}{}, handle.Write([]byte(p.Data))
}

func (h *TerminalHandlers) Resize(_ context.Context, raw json.RawMessage) (any, error) {
	var p TerminalResizeParams
	if err := json.Unmarshal(raw, &p); err != nil {
		return nil, err
	}
	h.mu.Lock()
	handle, ok := h.handles[p.TerminalID]
	h.mu.Unlock()
	if !ok {
		return nil, ErrSessionNotFound
	}
	return struct{}{}, handle.Resize(p.Cols, p.Rows)
}

func (h *TerminalHandlers) Close(ctx context.Context, raw json.RawMessage) (any, error) {
	var p TerminalCloseParams
	if err := json.Unmarshal(raw, &p); err != nil {
		return nil, err
	}
	h.mu.Lock()
	handle, ok := h.handles[p.TerminalID]
	if ok {
		delete(h.handles, p.TerminalID)
	}
	h.mu.Unlock()
	if !ok {
		return struct{}{}, nil
	}
	return struct{}{}, handle.Close(ctx)
}

func (h *TerminalHandlers) pumpData(tid string, handle pty.Handle) {
	for chunk := range handle.Data() {
		h.push.Push("terminal.data", TerminalDataEvent{
			TerminalID: tid, Data: string(chunk),
		})
	}
}

func (h *TerminalHandlers) pumpExit(tid string, handle pty.Handle) {
	info := <-handle.Exit()
	h.mu.Lock()
	delete(h.handles, tid)
	h.mu.Unlock()
	h.push.Push("terminal.exit", TerminalExitEvent{
		TerminalID: tid, Code: info.Code,
		Reason: string(info.Reason), Msg: info.Msg,
	})
}
```

```go
				if err := conn.WriteJSON(msg); err != nil { return }
			}

				"method": "terminal.data",
```

```go
// Package remote routes pty.Handle through agentred's JSON-RPC-over-WS
// terminal.* methods. The Client manages the ws connection lifecycle and
// fans push events out to per-Handle channels.
package remote

import (
	"context"
	"encoding/json"
	"errors"
	"sync"
	"sync/atomic"

	"github.com/gorilla/websocket"

	"agentre/internal/pkg/pty"
)

type SpawnRequest struct {
	SessionID  int64
	Cwd, Shell string
	Env        []string
	Cols, Rows uint16
}

type Client struct {
	conn  *websocket.Conn
	mu    sync.Mutex
	seq   atomic.Int64
	pending map[int64]chan rpcResponse

	handlesMu sync.Mutex
	handles   map[string]*remoteHandle
	closed    chan struct{}
}

type rpcResponse struct {
	Result json.RawMessage
	Error  string
}

type wsFrame struct {
	ID     *int64          `json:"id,omitempty"`
	Method string          `json:"method,omitempty"`
	Params json.RawMessage `json:"params,omitempty"`
	Result json.RawMessage `json:"result,omitempty"`
	Error  string          `json:"error,omitempty"`
}

func Dial(ctx context.Context, url string) (*Client, error) {
	c, _, err := websocket.DefaultDialer.DialContext(ctx, url, nil)
	if err != nil { return nil, err }
	cl := &Client{
		conn: c,
		pending: map[int64]chan rpcResponse{},
		handles: map[string]*remoteHandle{},
		closed: make(chan struct{}),
	}
	go cl.readLoop()
	return cl, nil
}

func (c *Client) Stop() {
	select {
	case <-c.closed:
	default:
		close(c.closed)
	}
	_ = c.conn.Close()
	c.handlesMu.Lock()
	for _, h := range c.handles {
		h.fireExit(pty.ExitInfo{Reason: pty.ExitConnectionLost, Msg: "client stopped"})
	}
	c.handlesMu.Unlock()
}

func (c *Client) readLoop() {
	for {
		var f wsFrame
		if err := c.conn.ReadJSON(&f); err != nil {
			c.handlesMu.Lock()
			for _, h := range c.handles {
				h.fireExit(pty.ExitInfo{Reason: pty.ExitConnectionLost, Msg: err.Error()})
			}
			c.handles = map[string]*remoteHandle{}
			c.handlesMu.Unlock()
			return
		}
		if f.ID != nil {
			c.mu.Lock()
			ch, ok := c.pending[*f.ID]
			delete(c.pending, *f.ID)
			c.mu.Unlock()
			if ok {
				ch <- rpcResponse{Result: f.Result, Error: f.Error}
			}
			continue
		}
		switch f.Method {
		case "terminal.data":
			var ev struct{ TerminalID, Data string }
			_ = json.Unmarshal(f.Params, &ev)
			c.handlesMu.Lock()
			h := c.handles[ev.TerminalID]
			c.handlesMu.Unlock()
			if h != nil {
				select {
				case h.data <- []byte(ev.Data):
				default: // drop on backpressure
				}
			}
		case "terminal.exit":
			var ev struct {
				TerminalID, Reason, Msg string
				Code                    int
			}
			_ = json.Unmarshal(f.Params, &ev)
			c.handlesMu.Lock()
			h := c.handles[ev.TerminalID]
			delete(c.handles, ev.TerminalID)
			c.handlesMu.Unlock()
			if h != nil {
				h.fireExit(pty.ExitInfo{Code: ev.Code, Reason: pty.ExitReason(ev.Reason), Msg: ev.Msg})
			}
		}
	}
}

func (c *Client) call(ctx context.Context, method string, params any) (json.RawMessage, error) {
	id := c.seq.Add(1)
	rawParams, _ := json.Marshal(params)
	ch := make(chan rpcResponse, 1)
	c.mu.Lock()
	c.pending[id] = ch
	c.mu.Unlock()

	if err := c.conn.WriteJSON(wsFrame{ID: &id, Method: method, Params: rawParams}); err != nil {
		return nil, err
	}
	select {
	case resp := <-ch:
		if resp.Error != "" {
			return nil, errors.New(resp.Error)
		}
		return resp.Result, nil
	case <-ctx.Done():
		return nil, ctx.Err()
	}
}

func (c *Client) Open(ctx context.Context, req SpawnRequest) (pty.Handle, error) {
	raw, err := c.call(ctx, "terminal.open", map[string]any{
		"sessionId": req.SessionID, "cwd": req.Cwd, "shell": req.Shell,
		"env": req.Env, "cols": req.Cols, "rows": req.Rows,
	})
	if err != nil { return nil, err }

	var res struct{ TerminalID string `json:"terminalId"` }
	if err := json.Unmarshal(raw, &res); err != nil { return nil, err }

	h := &remoteHandle{
		client: c, tid: res.TerminalID,
		data: make(chan []byte, 64),
		exit: make(chan pty.ExitInfo, 1),
	}
	c.handlesMu.Lock()
	c.handles[res.TerminalID] = h
	c.handlesMu.Unlock()
	return h, nil
}

type remoteHandle struct {
	client   *Client
	tid      string
	data     chan []byte
	exit     chan pty.ExitInfo
	exitOnce sync.Once
}

func (h *remoteHandle) Data() <-chan []byte       { return h.data }
func (h *remoteHandle) Exit() <-chan pty.ExitInfo { return h.exit }

func (h *remoteHandle) Write(data []byte) error {
	_, err := h.client.call(context.Background(), "terminal.write", map[string]any{
		"terminalId": h.tid, "data": string(data),
	})
	return err
}

func (h *remoteHandle) Resize(cols, rows uint16) error {
	_, err := h.client.call(context.Background(), "terminal.resize", map[string]any{
		"terminalId": h.tid, "cols": cols, "rows": rows,
	})
	return err
}

func (h *remoteHandle) Close(ctx context.Context) error {
	_, err := h.client.call(ctx, "terminal.close", map[string]any{"terminalId": h.tid})
	h.fireExit(pty.ExitInfo{Reason: pty.ExitKilled})
	return err
}

func (h *remoteHandle) fireExit(info pty.ExitInfo) {
	h.exitOnce.Do(func() {
		h.exit <- info
		close(h.exit)
		close(h.data)
	})
}
```

```go
func (h *TerminalHandlers) pumpData(tid string, handle pty.Handle) {
	const bufLimit = 256
	queue := make(chan []byte, bufLimit)
	throttled := false
	go func() {
		for chunk := range queue {
			h.push.Push("terminal.data", TerminalDataEvent{TerminalID: tid, Data: string(chunk)})
		}
	}()
	for chunk := range handle.Data() {
		select {
		case queue <- chunk:

		default:

				case queue <- []byte("\r\n[--- output throttled ---]\r\n"):

				}
				throttled = true
			}
		}
	}
	close(queue)
}
```
